// 加進 changchih-class 專案現有 Firestore 規則的 match /databases/{database}/documents { ... } 區塊內
// 只開放本簡報的開場投票集合：任何人可讀、可新增，不可修改或刪除；欄位長度有上限
match /tve1966_open/{id} {
  allow read: if true;
  allow create: if request.resource.data.keys().hasOnly(['choice','ts'])
                && request.resource.data.choice in ['普通高中','高職','五專','就業'];
}

// 1986 決策題：選項 A/B/C＋選填理由（60 字內），只能新增
match /tve1966_policy/{id} {
  allow read: if true;
  allow create: if request.resource.data.keys().hasOnly(['choice','reason','ts'])
                && request.resource.data.choice in ['A','B','C']
                && request.resource.data.reason is string
                && request.resource.data.reason.size() <= 60
                && request.resource.data.ts == request.time;
}
